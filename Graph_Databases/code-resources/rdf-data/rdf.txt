---------------- SPARQL query -------------------

PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
PREFIX ex: <http://example.org/>

SELECT ?customer ?blood ?passport ?item ?allergy
WHERE {
  ?customer a ex:Customer ;
            ex:hasMedicalRecord ?med ;
            ex:hasGovernmentRecord ?gov ;
            ex:hasPurchaseRecord ?pur .

  ?med ex:bloodGroup ?blood ;
       ex:allergy ?allergy .

  ?gov ex:passportNumber ?passport .

  ?pur ex:purchasedItem ?item .
}

# 1. Get the customer who purchased a laptop and has blood group B+
SELECT ?customer ?name
WHERE {
  ?customer a ex:Customer ;
            ex:name ?name ;
            ex:hasPurchaseRecord ?purchase ;
            ex:hasMedicalRecord ?med .

  ?purchase ex:purchasedItem "Laptop"^^xsd:string .
  ?med ex:bloodGroup "B+"^^xsd:string .
}


# 2. Get customers who purchased a laptop after 2025-01-14
SELECT ?customer ?name ?purchasedDate
WHERE {
  ?customer a ex:Customer ;
            ex:name ?name ;
            ex:hasPurchaseRecord ?purchase .

  ?purchase ex:purchasedItem "Laptop"^^xsd:string ;
            ex:purchasedDate ?purchasedDate .

  FILTER ( ?purchasedDate > "2025-01-14"^^xsd:date )
}
ORDER BY ?purchasedDate 

# 3. Who has “Penicillin” allergy and purchased a laptop?

    ?customer a ex:Customer;

                ex:hasPurchaseRecord ?purchase ;
                ex:hasMedicalRecord ?med .
    
    ?med ex:allergy "Penicillin"^^xsd:string .
    ?purchase ex:purchasedItem "Iphone"^^xsd:string .
    
    
}
